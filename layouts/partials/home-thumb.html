{{- /* Home thumbnail (plan 0008 / D7). 16 by 9 centre crop, WebP 480 by 270. Source: front matter `thumb` (optional override image, used when a cover crops badly), else `cover.image`. Posts with neither, or whose file is missing, get a token-coloured tile. Static images are mounted read-only at assets/thumbsrc (see [module] mounts in hugo.toml) so Hugo can resize them without moving them. Context: dict "page" . "eager" bool */ -}} {{- $page := .page }} {{- $src := $page.Params.thumb | default $page.Params.cover.image | default "" }} {{- $img := "" }} {{- if $src }}{{ $img = resources.Get (printf "thumbsrc/%s" (path.Base $src)) }}{{ end }} {{- /* Animated GIF covers cannot be encoded to WebP, so Fill on one fails the whole build. If the post ships a same-name still next to it (-final.png, then .png), thumb from that; otherwise serve the GIF untouched. Non-GIF sources take the existing Fill path unchanged. */ -}} {{- $still := "" }} {{- if and $img (eq $img.MediaType.SubType "gif") }} {{- $base := path.Base $src }} {{- $stem := strings.TrimSuffix (path.Ext $base) $base }} {{- range slice (printf "thumbsrc/%s-final.png" $stem) (printf "thumbsrc/%s.png" $stem) }} {{- if not $still }}{{ $still = resources.Get . }}{{ end }} {{- end }} {{- end }} {{- if $img }} {{- $sized := true }} {{- $t := $img }} {{- if $still }} {{- $t = $still.Fill "480x270 webp q80 Center" }} {{- else if eq $img.MediaType.SubType "gif" }} {{- $sized = false }} {{- else }} {{- $t = $img.Fill "480x270 webp q80 Center" }} {{- end }} {{- else }} {{ substr $page.Title 0 1 }} {{- end }}
