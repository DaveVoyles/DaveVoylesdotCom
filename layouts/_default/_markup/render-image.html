{{- /* Wrap markdown images so readers can click to enlarge (dialog lightbox). Title "sm" or "sm|" adds post-figure--sm (narrow display width). */ -}} {{- $u := .Destination | safeURL -}} {{- $alt := .Text | default "" -}} {{- $title := .Title | default "" -}} {{- $sm := false -}} {{- if eq $title "sm" -}} {{- $sm = true -}} {{- $title = "" -}} {{- else if strings.HasPrefix $title "sm|" -}} {{- $sm = true -}} {{- $title = strings.TrimPrefix "sm|" $title -}} {{- end -}}
